\begin{enumerate}
\item (6 pts) Determine the solution to the initial value problem
	\[2y''-3y'+y=0;\quad y(0)=2, \quad y'(0)=\frac{1}{2}.\]
\vspace{6cm}
\item (7 pts) Determine the solution to the initial value problem
	\[y''+2y'+2y=0;\quad y(\pi/4)=2, \quad y'(\pi/4)=-2.\]
\newpage
\item (7 pts) Determine the solution to the initial value problem (notice that the answer will be in terms of the constant b)
	\[y''-y'+\frac{y}{4}=0;\quad y(0)=2, \quad y'(0)=b.\]
\end{enumerate}
