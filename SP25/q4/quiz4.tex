\begin{enumerate}

	\item (20 points) Row reduce the following augmented matrices; after
		that, indicate if the associated system has zero, one or
		infinitely many solutions and explain why. Show your work and
		row reduce as much as possible, \textbf{do not} use backwards
		substitution. You do not need to solve the system. Just row
		reduce and then write if the number of solutions is 0, 1 or
		$\infty$ and a brief explanation of why.

		\begin{enumerate}

			\item

				$\begin{amatrix}{2}
				1 & 2 & 3\\
				3 & 2 & 3\\
				\end{amatrix}$


			\item 

				$\begin{amatrix}{3}
				0 & 2 & 3 & 0\\
				1 & 2 & 3 & 1\\
				2 & 2 & 3 & 1\\
				\end{amatrix}$

			\item

				$\begin{amatrix}{3}
				0 & 2 & 3 & 1\\
				1 & 2 & 3 & 1\\
				2 & 2 & 3 & 1\\
				\end{amatrix}$

			\item

				$\begin{amatrix}{4}
				6 & 3 & 12 & 0 & 3\\
				1 & 2 & 3 & 1 & 2\\
				0 & 3 & 2 & 2 & 3\\
				2 & 1 & 4 & 0 & 1\\
				\end{amatrix}$

		\end{enumerate}

\end{enumerate}
